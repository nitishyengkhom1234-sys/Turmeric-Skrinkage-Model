  "temperature": 80.0,
  "time": 15.0
}

Response Schema (200 OK)

Integration Examples

curl -X POST https://your-domain.vercel.app/api/predict \
  -H "Content-Type: application/json" \
  -d '{"temperature": 80.0, "time": 15.0}'
import requests

url = "https://your-domain.vercel.app/api/predict"
payload = {"temperature": 80.0, "time": 15.0}
response = requests.post(url, json=payload)
data = response.json()

print(f"Predicted Shrinkage Ratio: {data['predicted_shrinkage_ratio']}")
const res = await fetch('/api/predict', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({ temperature: 80.0, time: 15.0 })
});
const data = await res.json();
console.log('Prediction:', data.predicted_shrinkage_ratio);
