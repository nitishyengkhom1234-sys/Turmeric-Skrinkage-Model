Production REST API Reference (Unit-Aware)

Serverless endpoints supporting optional units (°C, °F, K, min, hr, s)

POST /api/predict

Request Schema (Optional Units)

{
  "temperature": 176.0,
  "temp_unit": "F",       // "C", "F", or "K" (default "C")
  "time": 0.25,
  "time_unit": "hr"       // "min", "hr", or "s" (default "min")
}

Response Schema (200 OK)

{
  "input": {
    "temperature": 176.0,
    "temp_unit": "F",
    "time": 0.25,
    "time_unit": "hr"
  },
  "normalized_for_model": {
    "temperature_celsius": 80.0,
    "time_minutes": 15.0
  },
  "predicted_shrinkage_ratio": 0.3593,
  "volume_reduction_percent": 64.07,
  "model": "MLPRegressor (8, 4)",
  "r2_score": 0.99661
}

Integration Examples

curl -X POST https://turmeric-skrinkage-model.vercel.app/api/predict \
  -H "Content-Type: application/json" \
  -d '{"temperature": 176.0, "temp_unit": "F", "time": 0.25, "time_unit": "hr"}'
import requests

url = "https://turmeric-skrinkage-model.vercel.app/api/predict"
payload = {
    "temperature": 176.0,
    "temp_unit": "F",
    "time": 0.25,
    "time_unit": "hr"
}
response = requests.post(url, json=payload)
data = response.json()

print(f"Predicted Shrinkage Ratio: {data['predicted_shrinkage_ratio']}")
print(f"Model Input (°C, min): {data['normalized_for_model']}")
const res = await fetch('/api/predict', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({
    temperature: 176.0,
    temp_unit: 'F',
    time: 0.25,
    time_unit: 'hr'
  })
});
const data = await res.json();
console.log('Prediction:', data.predicted_shrinkage_ratio);
